+### APFS clone workflow
+
+Every `_keep`, `_aside`, and `_level-XXX` materialization runs through `src/fs/copyOps.js`, which:
+
+- Prefers APFS clonefile operations on macOS when source and destination share a volume.
+- Falls back to same-device hardlinks, then metadata-preserving byte copies.

+- Emits a JSON log line per file to `data/journal.ndjson` and prints a summary (mode counts, elapsed wall clock, `du -sk` delta).

+async function performCopy(src, dst, strategy, options = {}) {
+  const { dryRun = false, journalPath } = options;
+  const effectiveStrategy = normalizeStrategy(
+    options.strategy ?? process.env.COPY_STRATEGY ?? strategy
+  );
+  const entry = { op: "copy", from: src, to: dst };
+  const start = Date.now();
+  const srcStat = await fs.lstat(src);
+  if (!srcStat.isFile() && !srcStat.isSymbolicLink()) {
+    const result = { mode: "skip", bytes: 0, ms: msSince(start) };
+    logOperation({ ...entry, mode: "skip", bytes: 0, ms: result.ms }, { journalPath });
+    updateMetrics({ mode: "skip", bytes: 0 });
+    return result;
+  }
+  if (srcStat.isSymbolicLink()) {
+    return applySymlink(src, dst, entry, options);
+  }
+  await fs.mkdir(path.dirname(dst), { recursive: true });
+  const tmpPath = pickTmpPath(dst);
+  const existing = await statMaybe(dst);
+  if (existing && !dryRun) {
+    // Idempotency: skip if file already exists with same size and mtime.
+    if (
+      existing.isFile() &&
+      existing.size === srcStat.size &&
+      Math.abs(existing.mtimeMs - srcStat.mtimeMs) < 5
+    ) {
+      const result = { mode: "skip", bytes: 0, ms: msSince(start) };
+      logOperation({ ...entry, mode: "skip", bytes: 0, ms: result.ms }, { journalPath });
+      updateMetrics({ mode: "skip", bytes: 0 });
+      return result;
+    }
+    await fs.unlink(dst);
+  }
+
+  const destDirStat = await fs.stat(path.dirname(dst));
+  const sameDevice = destDirStat.dev === srcStat.dev;
+  const apfs = sameDevice ? await isApfs(path.dirname(dst)) : false;
+
+  const order = resolveModeOrder(effectiveStrategy, { sameDevice, apfs });
+  let mode = "copy";
+  let success = false;
+  let error = null;
+  try {
+    if (dryRun) {
+      mode = order[0] || "copy";
+      success = true;
+    } else {
+      for (const candidate of order) {
+        if (candidate === "clone") {
+          if (!sameDevice) continue;
+          let cloneMode = await tryClone(src, tmpPath);
+          if (!cloneMode && apfs) cloneMode = await tryCopyfileClone(src, tmpPath);
+          if (!cloneMode && apfs) cloneMode = await tryCpClone(src, tmpPath);
+          if (cloneMode) {
+            mode = "clone";
+            success = true;
+            break;
+          }
+        } else if (candidate === "hardlink") {
+          if (!sameDevice) continue;
+          const hardlinked = await tryHardlink(src, tmpPath);
+          if (hardlinked) {
+            mode = "hardlink";
+            success = true;
+            break;
+          }
+        } else if (candidate === "copy") {
+          mode = await metadataCopy(src, tmpPath, srcStat);
+          success = true;
+          break;
+        }
+      }
+      if (!success) {
+        mode = await metadataCopy(src, tmpPath, srcStat);
+        success = true;
+      }
+      await fs.rename(tmpPath, dst);
+    }
+  } catch (err) {
+    error = err;
+    success = false;

+        // ignore cleanup errors otherwise
+      }
+    }
+  }

+        mode: "error",

+        error: error?.message,
+        code: error?.code,
+      },
+      { journalPath }

+    throw error;
+  }

+async function walkTree(srcDir) {
+  const directories = [];
+  const files = [];
+  const stack = [{ src: srcDir, dst: null }];
+  while (stack.length) {
+    const { src } = stack.pop();
+    const entries = await fs.readdir(src, { withFileTypes: true });
+    for (const entry of entries) {
+      const srcPath = path.join(src, entry.name);
+      const rel = path.relative(srcDir, srcPath);
+      const info = { src: srcPath, rel, entry };
+      if (entry.isDirectory()) {
+        directories.push(info);
+        stack.push({ src: srcPath, dst: null });
+      } else {
+        files.push(info);
+      }
+    }
+  }
+  directories.sort((a, b) => a.rel.localeCompare(b.rel));
+  files.sort((a, b) => a.rel.localeCompare(b.rel));
+  return { directories, files };
+}

+async function withLimit(items, limit, fn) {
+  const queue = items.slice();
+  const workers = Array.from({ length: Math.min(limit, queue.length) }, () =>
+    (async function worker() {
+      while (queue.length) {
+        const item = queue.shift();
+        await fn(item);
+      }
+    })()
+  );
+  await Promise.all(workers);
+}

+describe("copyOps", () => {
+  it("copies a file and skips when rerun", async () => {
+    const src = path.join(tmpDir, "source.txt");
+    const dst = path.join(tmpDir, "dest.txt");

+    const contents = await fs.readFile(dst, "utf8");
+    expect(contents).toBe("hello world");
+    const second = await copyFile(src, dst, "auto", { journalPath });
+    expect(second.mode === "skip" || second.mode === first.mode).toBe(true);
+    const summary = await summarizeCopySession({ roots: [tmpDir] });
+    expect(summary.counts).toBeDefined();
